SQL ติดตั้งฐานข้อมูล Supabase PostgreSQL & RLS

-- Supabase PostgreSQL Schema for diginae.cam
create table if not exists public.cameras (
  id text primary key,
  name text not null,
  description text,
  image text,
  gallery jsonb default '[]'::jsonb,
  hourly_tiers jsonb default '[]'::jsonb,
  daily_tiers jsonb default '[]'::jsonb,
  deposit integer default 1000,
  units jsonb default '[]'::jsonb
);

create table if not exists public.bookings (
  code text primary key,
  camera_id text references public.cameras(id),
  unit_code text,
  pickup_date date not null,
  pickup_time text not null,
  return_date date not null,
  return_time text not null,
  rent_price integer not null,
  deposit integer not null,
  total_amount integer not null,
  pickup_location text not null,
  renter_name text not null,
  renter_phone text not null,
  status text default 'รอตรวจสอบ',
  created_at timestamptz default now()
);

alter table public.cameras enable row level security;
alter table public.bookings enable row level security;
